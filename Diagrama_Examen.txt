Enum estado_cliente {
  ACTIVO
  INACTIVO
}

Enum estado_envio {
  REGISTRADO
  EN_TRANSITO
  ENTREGADO
  CANCELADO
}

Table clientes {
  id int [pk, increment]
  nombre varchar(100)
  estado estado_cliente
  limite_credito decimal(10,2)
}

Table direcciones {
  id int [pk, increment]
  cliente_id int [not null, ref: > clientes.id]
  direccion varchar(255)
}

Table envios {
  id int [pk, increment]
  remitente_id int [not null, ref: > clientes.id]
  destinatario_id int [not null, ref: > clientes.id]
  direccion_origen_id int [not null, ref: > direcciones.id]
  direccion_destino_id int [not null, ref: > direcciones.id]
  costo_total decimal(10,2)
  fecha datetime
  estado estado_envio
}

Table almacenes {
  id int [pk, increment]
  nombre varchar(100)
  ubicacion varchar(255)
  capacidad_maxima int
}

Table paquetes {
  id int [pk, increment]
  envio_id int [not null, ref: > envios.id]
  almacen_id int [not null, ref: > almacenes.id]
  peso decimal(10,2)
  dimensiones varchar(50)
  contenido varchar(255)
}

Table historial_rastreo {
  id int [pk, increment]
  envio_id int [not null, ref: > envios.id]
  fecha datetime
  ubicacion_actual varchar(255)
  estado_evento estado_envio
  observaciones text
}

Table auditoria_envios {
  id int [pk, increment]
  envio_id int [not null, ref: > envios.id]
  costo_anterior decimal(10,2)
  costo_nuevo decimal(10,2)
  usuario varchar(100)
  fecha_modificacion datetime
}
